def init_db():
    os.makedirs('data', exist_ok=True)
    conn = sqlite3.connect('data/database.db', timeout=20)
    c = conn.cursor()
    
    c.execute('PRAGMA foreign_keys = ON')
    
    # Пользователи
    c.execute('''CREATE TABLE IF NOT EXISTS users
                 (id INTEGER PRIMARY KEY AUTOINCREMENT,
                  telegram_id INTEGER UNIQUE,
                  username TEXT,
                  first_name TEXT,
                  balance INTEGER DEFAULT 10000,
                  avatar_color TEXT DEFAULT '#9b59b6',
                  avatar_path TEXT DEFAULT NULL,
                  banned INTEGER DEFAULT 0,
                  tester INTEGER DEFAULT 0,
                  tester_lvl INTEGER DEFAULT 0,
                  tester_balance INTEGER DEFAULT 0,
                  frozen INTEGER DEFAULT 0,
                  login_key TEXT UNIQUE,
                  created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
                  last_login TIMESTAMP,
                  verified INTEGER DEFAULT 0,
                  verified_by TEXT DEFAULT NULL,
                  premium_until TIMESTAMP DEFAULT NULL,
                  premium_forever INTEGER DEFAULT 0)''')
    
    # Подарки
    c.execute('''CREATE TABLE IF NOT EXISTS gifts
                 (id INTEGER PRIMARY KEY AUTOINCREMENT,
                  name TEXT,
                  price INTEGER,
                  image TEXT,
                  total_count INTEGER DEFAULT -1,
                  sold_count INTEGER DEFAULT 0,
                  per_user_limit INTEGER DEFAULT -1,
                  created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
                  is_upgrade INTEGER DEFAULT 0,
                  available INTEGER DEFAULT 1,
                  is_tester_only INTEGER DEFAULT 0,
                  premium_only INTEGER DEFAULT 0,
                  available_time TIMESTAMP DEFAULT NULL)''')
    
    # Инвентарь (без collection_number)
    c.execute('''CREATE TABLE IF NOT EXISTS inventory
                 (id INTEGER PRIMARY KEY AUTOINCREMENT,
                  user_id INTEGER,
                  gift_id INTEGER,
                  background TEXT DEFAULT 'none',
                  upgraded_at TIMESTAMP,
                  upgrade_image TEXT,
                  FOREIGN KEY (user_id) REFERENCES users (id) ON DELETE CASCADE,
                  FOREIGN KEY (gift_id) REFERENCES gifts (id) ON DELETE CASCADE)''')
    
    # Рынок
    c.execute('''CREATE TABLE IF NOT EXISTS market
                 (id INTEGER PRIMARY KEY AUTOINCREMENT,
                  user_id INTEGER,
                  inventory_id INTEGER,
                  price INTEGER,
                  listed_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
                  is_sold INTEGER DEFAULT 0,
                  FOREIGN KEY (user_id) REFERENCES users (id),
                  FOREIGN KEY (inventory_id) REFERENCES inventory (id))''')
    
    # Изображения улучшений
    c.execute('''CREATE TABLE IF NOT EXISTS upgrade_images
                 (id INTEGER PRIMARY KEY AUTOINCREMENT,
                  gift_id INTEGER,
                  image_path TEXT,
                  created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
                  FOREIGN KEY (gift_id) REFERENCES gifts (id) ON DELETE CASCADE)''')
    
    # Покупки
    c.execute('''CREATE TABLE IF NOT EXISTS purchases
                 (id INTEGER PRIMARY KEY AUTOINCREMENT,
                  user_id INTEGER,
                  gift_id INTEGER,
                  quantity INTEGER,
                  purchased_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
                  FOREIGN KEY (user_id) REFERENCES users (id),
                  FOREIGN KEY (gift_id) REFERENCES gifts (id))''')
    
    # Настройки сайта
    c.execute('''CREATE TABLE IF NOT EXISTS site_settings
                 (id INTEGER PRIMARY KEY AUTOINCREMENT,
                  whitelist_mode INTEGER DEFAULT 0,
                  market_enabled INTEGER DEFAULT 1,
                  chat_enabled INTEGER DEFAULT 1,
                  chat_name TEXT DEFAULT 'Общий чат',
                  chat_avatar TEXT DEFAULT NULL,
                  created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP)''')
    
    # Сообщения чата
    c.execute('''CREATE TABLE IF NOT EXISTS chat_messages
                 (id INTEGER PRIMARY KEY AUTOINCREMENT,
                  user_id INTEGER,
                  message TEXT,
                  created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
                  is_deleted INTEGER DEFAULT 0,
                  FOREIGN KEY (user_id) REFERENCES users (id) ON DELETE CASCADE)''')
    
    # Топ по времени
    c.execute('''CREATE TABLE IF NOT EXISTS daily_top
                 (id INTEGER PRIMARY KEY AUTOINCREMENT,
                  user_id INTEGER,
                  date DATE DEFAULT CURRENT_DATE,
                  claimed INTEGER DEFAULT 0,
                  FOREIGN KEY (user_id) REFERENCES users (id))''')
    
    c.execute('CREATE INDEX IF NOT EXISTS idx_users_telegram_id ON users(telegram_id)')
    c.execute('CREATE INDEX IF NOT EXISTS idx_users_login_key ON users(login_key)')
    c.execute('CREATE INDEX IF NOT EXISTS idx_inventory_user_id ON inventory(user_id)')
    c.execute('CREATE INDEX IF NOT EXISTS idx_chat_messages_created_at ON chat_messages(created_at)')
    
    c.execute('INSERT OR IGNORE INTO site_settings (id, whitelist_mode, market_enabled, chat_enabled, chat_name) VALUES (1, 0, 1, 1, "Общий чат")')
    
    conn.commit()
    conn.close()
    print("База данных инициализирована!")

def generate_login_key():
    return ''.join(random.choices(string.ascii_letters + string.digits, k=16))

def create_or_get_user(telegram_id, username, first_name):
    conn = get_db()
    user = conn.execute('SELECT * FROM users WHERE telegram_id = ?', (telegram_id,)).fetchone()
    
    if not user:
        login_key = generate_login_key()
        conn.execute('''INSERT INTO users (telegram_id, username, first_name, avatar_color, balance, login_key) 
                        VALUES (?, ?, ?, ?, 10000, ?)''',
                     (telegram_id, username, first_name, '#9b59b6', login_key))
        conn.commit()
        user = conn.execute('SELECT * FROM users WHERE telegram_id = ?', (telegram_id,)).fetchone()
    else:
        if not user['login_key']:
            login_key = generate_login_key()
            conn.execute('UPDATE users SET login_key = ? WHERE telegram_id = ?', (login_key, telegram_id))
            conn.commit()
            user = conn.execute('SELECT * FROM users WHERE telegram_id = ?', (telegram_id,)).fetchone()
    
    user_dict = dict(user)
    conn.close()
    return user_dict

def is_admin(user_id):
    user = get_user_by_id(user_id)
    return user and user['telegram_id'] in ADMIN_IDS

def is_tester(user_id):
    user = get_user_by_id(user_id)
    return user and user['tester'] == 1

def is_frozen(user_id):
    user = get_user_by_id(user_id)
    return user and user['frozen'] == 1

def has_premium(user_id):
    user = get_user_by_id(user_id)
    if not user:
        return False
    if user['premium_forever'] == 1:
        return True
    if user['premium_until']:
        return datetime.now() < datetime.strptime(user['premium_until'], '%Y-%m-%d %H:%M:%S')
    return False

def login_required(f):
    @wraps(f)
    def decorated(*args, **kwargs):
        if 'user_id' not in session:
            return redirect('/')
        user = get_user_by_id(session['user_id'])
        if not user or user['banned'] == 1 or user['frozen'] == 1:
            session.clear()
            return redirect('/')
        return f(*args, **kwargs)
    return decorated

def admin_required(f):
    @wraps(f)
    def decorated(*args, **kwargs):
        if 'user_id' not in session:
            return redirect('/')
        if not is_admin(session['user_id']):
            return "Доступ запрещен", 403
        return f(*args, **kwargs)
    return decorated

def get_all_gifts():
    conn = get_db()
    gifts = conn.execute('''SELECT *, 
                            CASE 
                                WHEN total_count != -1 AND sold_count >= total_count THEN 1 
                                ELSE 0 
                            END as is_sold_out
                          FROM gifts 
                          WHERE available = 1 
                          ORDER BY is_sold_out, created_at DESC''').fetchall()
    conn.close()
    return [dict(g) for g in gifts]

def get_user_inventory(user_id):
    conn = get_db()
    items = conn.execute('''SELECT inventory.*, gifts.name, gifts.image, gifts.is_upgrade,
                                   market.id as market_id,
                                   CASE 
                                     WHEN inventory.upgrade_image IS NOT NULL THEN inventory.upgrade_image
                                     WHEN gifts.image IS NOT NULL AND gifts.image != '' THEN gifts.image
                                     ELSE 'static/default-gift.png'
                                   END as display_image
                           FROM inventory 
                           LEFT JOIN gifts ON inventory.gift_id = gifts.id
                           LEFT JOIN market ON inventory.id = market.inventory_id AND market.is_sold = 0
                           WHERE inventory.user_id = ?
                           ORDER BY inventory.id DESC''', (user_id,)).fetchall()
    conn.close()
    return [dict(i) for i in items]

def get_chat_messages(limit=50):
    conn = get_db()
    messages = conn.execute('''SELECT chat_messages.*, users.username, users.first_name, users.avatar_path, users.avatar_color,
                                      users.verified, users.verified_by, users.tester, users.tester_lvl
                              FROM chat_messages 
                              JOIN users ON chat_messages.user_id = users.id
                              WHERE chat_messages.is_deleted = 0
                              ORDER BY chat_messages.created_at DESC 
                              LIMIT ?''', (limit,)).fetchall()
    conn.close()
    return [dict(m) for m in messages][::-1]

def send_chat_message(user_id, message):

def check_daily_top():
    conn = get_db()
    today = datetime.now().date()
    
    # Проверяем, получал ли кто-то сегодня награду
    claimed = conn.execute('SELECT user_id FROM daily_top WHERE date = ? AND claimed = 1', (today,)).fetchone()
    
    if not claimed:
        # Находим топ-1 по балансу
        top_user = conn.execute('SELECT id FROM users WHERE banned = 0 AND frozen = 0 ORDER BY balance DESC LIMIT 1').fetchone()
        if top_user:
            # Начисляем 20000
            conn.execute('UPDATE users SET balance = balance + 20000 WHERE id = ?', (top_user['id'],))
            conn.execute('INSERT INTO daily_top (user_id, date, claimed) VALUES (?, ?, 1)', (top_user['id'], today))
            conn.commit()
            
            # Уведомляем в бота
            try:
                user = get_user_by_id(top_user['id'])
                if user and user['telegram_id']:
                    bot.send_message(user['telegram_id'], "🏆 Вы стали топом дня! +20 000 звёзд!")
            except:
                pass
    
    conn.close()
